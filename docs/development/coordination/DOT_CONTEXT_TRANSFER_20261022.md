import os
import json
import time
import selectors
import subprocess
from datetime import datetime
from zoneinfo import ZoneInfo

profiles = [
    (
        "MAIN",
        "/root/.codex",
        "/root/.nvm/versions/node/v22.22.2/bin/codex"
    ),
    (
        "CODEX2",
        "/root/.codex-profiles/codex2",
        "/root/.codex-profiles/codex2/packages/app-server-daemon/releases/0.163.0-x86_64-unknown-linux-musl/bin/codex"
    )
]

def format_reset(timestamp):
    if timestamp is None:
        return "неизвестно"
    return datetime.fromtimestamp(
        timestamp,
        ZoneInfo("Etc/GMT-5")
    ).strftime("%d.%m.%Y %H:%M:%S UTC+5")

def print_window(name, window):
    if not window:
        print(f"{name}: данные отсутствуют")
        return

    used = window["usedPercent"]
    remaining = 100 - used

    print(f"{name}:")
    print(f"  Использовано: {used}%")
    print(f"  Осталось: {remaining}%")
    print(f"  Продолжительность: {window.get('windowDurationMins')} минут")
    print(f"  Сброс: {format_reset(window.get('resetsAt'))}")

    print(f"\n========== {name} ==========", flush=True)

    env = os.environ.copy()
    env["CODEX_HOME"] = home

    process = subprocess.Popen(
        [executable, "app-server", "--stdio"],
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=False,
        bufsize=0,
        env=env
    )

    def send(message):
        process.stdin.write(
            (json.dumps(message, separators=(",", ":")) + "\n").encode("utf-8")
        )
        process.stdin.flush()

    selector = selectors.DefaultSelector()
    os.set_blocking(process.stdout.fileno(), False)
    selector.register(process.stdout, selectors.EVENT_READ)
    pending_lines = []
    buffer = b""

    try:
        send({
            "id": 1,
            "method": "initialize",
            "params": {
                "clientInfo": {
                    "name": "quota-readonly-check",
                    "version": "1.0.0"
                },
                "capabilities": {
                    "experimentalApi": True
                }
            }
        })

        deadline = time.monotonic() + 20
        completed = False

        while time.monotonic() < deadline:
            if not pending_lines:
                if not selector.select(timeout=0.5):
                    if process.poll() is not None:
                        break
                    continue
                try:

                if not chunk:

                if not pending_lines:

Выполнение команды на Easyscript

Codex App Server

### 5. Как интерпретировать ответ

- `primary.usedPercent` — процент израсходованного лимита в основном окне.
- `secondary.usedPercent` — израсходованный процент дополнительного окна, если оно есть.
- `windowDurationMins: 300` — пятичасовой лимит.
- `windowDurationMins: 10080` — недельный лимит.
- `resetsAt` — точная дата сброса в Unix timestamp.
- `ordinaryUsageAllowed` — разрешено ли обычное использование.
- `rateLimitResetCredits.availableCount` — число доступных бесплатных полных сбросов.

Остаток = 100 − usedPercent.

Если `secondary` отсутствует, нельзя считать его остаток равным 100%. Нужно писать «не предоставлено».
